proc ndr_test {args} {
    parse_proc_arguments -args $args result
    array set lw_map {}
    array set ls_map {}
    if {[info exists result(-layer_width)]} {
        if {[llength $result(-layer_width)] %2 !=0} {
            error "nomatch -layer_width {layer1 val1 layer2 val2 ...}"
        }
        array set lw_map $result(-layer_width)
    }
    if {[info exists result(-layer_space)]} {
        if {[llength $result(-layer_space)] %2 !=0} {
            error "nomatch -layer_space {layer1 val1 layer2 val2 ...}"
        }
        array set ls_map $result(-layer_space)
    }
    set width_list ""
    set space_list ""
    foreach layer $result(-layer) {
        if {[info exists lw_map($layer)]} {
            set w_coeff $lw_map($layer)
        } else {
            if {![info exists result(-width)]} {
                error "no width no layer_width"
            }
            set w_coeff $result(-width)
        }
        if {[info exists ls_map($layer)]} {
            set s_coeff $ls_map($layer)
        } else {
            if {![info exists result(-space)]} {
                error "no space no layer_space"
            }
            set s_coeff $result(-space)
        }
        set min_w [get_attribute [get_layer $layer] min_width]
        set pitch [get_attribute [get_layer $layer] pitch]
        set W($layer) [expr {$min_w * $w_coeff}]
        set S($layer) [expr {$pitch * $s_coeff - $W($layer)/2.0 - $min_w/2.0}]
        lappend width_list $layer
        lappend width_list $W($layer)
        lappend space_list $layer
        lappend space_list $S($layer)
    }
    create_routing_rule $result(-name)  -widths $width_list  -spacings $space_list
}

define_proc_attributes ndr_test \
    -info "INFO: create routing rule,support per layer width/space coeff or global coeff" \
    -define_args {
        {-name       ""  "routing rule name" string required}
        {-layer      ""  "layer list" list required}
        {-width      ""  "global width coeff (fallback if no -layer_width)" float optional}
        {-space      ""  "global space coeff (fallback if no -layer_space)" float optional}
        {-layer_width "" "per layer width coeff, format {M1 1.0 M2 1.5}" list optional}
        {-layer_space "" "per layer space coeff, format {M1 0.8 M2 2.0}" list optional}
    }
